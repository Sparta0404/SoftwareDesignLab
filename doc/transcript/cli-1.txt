# Phase 1 基础会话：编辑、撤销/重做、错误、文件往返（命令都是内置的；全装配下 tree 会多出 tree-star 的 " *"）
# 只含命令与注释；预期输出见 cli-1.expected.txt
init 800 600
create rect bg 0 0 800 600 #ffffff
create text title 20 40 "My Diagram"
create circle c1 100 100 20 red
create rect box 200 80 60 60 #3366ff
group markers c1 box
tree

# 编辑与撤销/重做
move c1 30 0
undo
redo
set title text "My Diagram v2"
undo
show c1

# 错误路径：非法分组、未知命令、未知属性、不存在的文件
group bad c1 markers
frobnicate
set c1 bogus 1
read missing-file.svg

# 删除、撤销，再保存；保存后不能 undo
delete box
undo
save demo.svg
undo

# 清空后读回保存点，文档内容恢复
delete markers
tree
read demo.svg
tree
help
